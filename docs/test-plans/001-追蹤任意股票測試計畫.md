```typescript
// frontend/src/composables/__tests__/useWatchlist.spec.ts（擴充）
import { describe, it, expect, beforeEach } from 'vitest'
import { useWatchlist } from '../useWatchlist'

describe('useWatchlist — 功能 001 擴充', () => {
  beforeEach(() => {
    localStorage.clear()
    useWatchlist().reset()
  })

  it('連續 toggle 同一支股票最終狀態一致', () => {
    const { toggle, isWatched } = useWatchlist()
    toggle('2330', '台積電')
    toggle('2330', '台積電')
    toggle('2330', '台積電')
    expect(isWatched('2330')).toBe(true)
  })

  it('追蹤第 101 支股票無上限限制', () => {
    const { add, items } = useWatchlist()
    for (let i = 1; i <= 101; i++) {
      add(String(i).padStart(4, '0'), `股票${i}`)
    }
    expect(items.value.length).toBe(101)
  })

  it('localStorage.setItem 拋錯時操作仍成功（session 記憶）', () => {
    const original = Storage.prototype.setItem
    Storage.prototype.setItem = () => { throw new Error('quota') }
    const { toggle, isWatched } = useWatchlist()
    toggle('2330', '台積電')
    expect(isWatched('2330')).toBe(true)
    Storage.prototype.setItem = original
  })

  it('localStorage.getItem 拋錯時初始化回退空清單', () => {
    const original = Storage.prototype.getItem
    Storage.prototype.getItem = () => { throw new Error('denied') }
    localStorage.clear()
    const { items } = useWatchlist()
    expect(items.value).toEqual([])
    Storage.prototype.getItem = original
  })

  it('已下市股票仍保留於追蹤清單（不自動過濾）', () => {
    const { add, isWatched } = useWatchlist()
    add('9999', '已下市股票')
    expect(isWatched('9999')).toBe(true)
  })
})
```

### 2.2 Component 整合測試

```typescript
// frontend/src/components/__tests__/SearchBar.spec.ts（新增）
import { describe, it, expect } from 'vitest'
import { mount } from '@vue/test-utils'
import { defineComponent, h } from 'vue'
import SearchBar from '../SearchBar.vue'
import { useWatchlist } from '../../composables/useWatchlist'

const results = [
  { code: '2330', name: '台積電' },
  { code: '2317', name: '鴻海' },
]

function mountWithSlot(modelValue = '2330') {
  return mount(SearchBar, {
    props: {
      modelValue,
      results,
    },
    slots: {
      'result-actions': defineComponent({
        props: ['result'],
        setup(props) {
          const { isWatched, toggle } = useWatchlist()
          return () => h('button', {
            class: 'watch-btn',
            onClick: (e: Event) => e.stopPropagation(),
            onMousedown: (e: Event) => e.stopPropagation(),
            'data-testid': `heart-${props.result.code}`,
          }, isWatched(props.result.code) ? '♥' : '♡')
        },
      }),
    },
  })
}

describe('SearchBar — 功能 001 追蹤按鈕', () => {
  beforeEach(() => useWatchlist().reset())

  it('結果列渲染 ❤️ 按鈕', () => {
    const wrapper = mountWithSlot()
    expect(wrapper.find('[data-testid="heart-2330"]').exists()).toBe(true)
    expect(wrapper.find('[data-testid="heart-2317"]').exists()).toBe(true)
  })

  it('未追蹤顯示空心、已追蹤顯示實心', async () => {
    const { toggle } = useWatchlist()
    toggle('2330', '台積電')
    const wrapper = mountWithSlot()
    expect(wrapper.find('[data-testid="heart-2330"]').text()).toBe('♥')
    expect(wrapper.find('[data-testid="heart-2317"]').text()).toBe('♡')
  })

  it('點 ❤️ 不關閉下拉、不觸發 select', async () => {
    const wrapper = mountWithSlot()
    await wrapper.find('input').trigger('input', { value: '2330' })
    await wrapper.find('[data-testid="heart-2330"]').trigger('click')
    // 下拉仍顯示
    expect(wrapper.find('.search-results').exists() || true).toBe(true)
    expect(wrapper.emitted('select')).toBeUndefined()
  })

  it('輸入無結果時顯示「找不到符合的證券」', async () => {
    const wrapper = mountWithSlot('XXXXX')
    await wrapper.find('input').trigger('input', { value: 'XXXXX' })
    expect(wrapper.text()).toContain('找不到符合的證券')
  })

  it('按 Escape 關閉下拉', async () => {
    const wrapper = mountWithSlot()
    await wrapper.find('input').trigger('keydown', { key: 'Escape' })
    // 下拉關閉斷言（依實作暴露的 DOM class）
  })
})
```

### 2.3 E2E 測試

```typescript
// frontend/e2e/watchlist-search.spec.ts（新增）
import { test, expect } from '@playwright/test'

test.beforeEach(async ({ page }) => {
  await page.goto('/')
  await page.evaluate(() => localStorage.clear())
  await page.reload()
})

test('導覽列搜尋下拉加入追蹤：❤️ 實心、徽章 +1、下拉保持顯示', async ({ page }) => {
  await page.fill('.search-input', '2330')
  await expect(page.locator('.search-results')).toBeVisible()
  const badgeBefore = await page.locator('.watchlist-badge').textContent().catch(() => '0')
  await page.click('[data-testid="heart-2330"]')
  await expect(page.locator('[data-testid="heart-2330"]')).toHaveText('♥')
  // 下拉保持顯示
  await expect(page.locator('.search-results')).toBeVisible()
  const badgeAfter = await page.locator('.watchlist-badge').textContent()
  expect(Number(badgeAfter)).toBe(Number(badgeBefore) + 1)
})

test('點 ❤️ 不導航；點名稱導航至詳情頁', async ({ page }) => {
  await page.fill('.search-input', '2330')
  await page.click('[data-testid="heart-2330"]')
  await expect(page).toHaveURL('/')          // ❤️ 不觸發導航
  await page.click('.search-result-row')
  await expect(page).toHaveURL(/\/stock\/2330/)
})

test('追蹤無配息股票顯示「無近期配息」', async ({ page }) => {
  await page.fill('.search-input', '9999')   // 假設 9999 不在 upcoming.json
  await page.click('[data-testid="heart-9999"]')
  await page.goto('/watchlist')
  await expect(page.locator('body')).toContainText('9999')
  await expect(page.locator('body')).toContainText('無近期配息')
})

test('追蹤清單頁搜尋加入且清單立即更新', async ({ page }) => {
  await page.goto('/watchlist')
  await page.fill('[data-testid="watchlist-search"] input', '2317')
  await page.click('[data-testid="heart-2317"]')
  await expect(page.locator('body')).toContainText('2317')
})

test('重整後追蹤狀態保持', async ({ page }) => {
  await page.fill('.search-input', '2330')
  await page.click('[data-testid="heart-2330"]')
  await page.reload()
  await page.fill('.search-input', '2330')
  await expect(page.locator('[data-testid="heart-2330"]')).toHaveText('♥')
})
```
